AddComponentPostInit("weapon", function(self)
	self.altattacking = false
	self.altattackfn = nil
	self.base_damage = self.damage
	self.alt_type = ""
	
	function self:SetOnAltAttack(fn)
		self.altattackfn = fn
	end
	
	function self:SetStimuli(stimuli)
		self.stimuli = stimuli
	end
	
	function self:SetBaseDamage(damage)
		self.base_damage = damage
		self.damage = damage
	end
	
	-- Need to set this if the alt does any form of damage
	function self:SetAltDamage(damage)
		self.alt_damage = damage
	end
	
	-- Need types for certain passives/buffs that only effect spells
	function self:SetAltType(alt_type)
		self.alt_type = alt_type
	end
	
	function self:AltAttack()
		self.altattacking = true
		self.damage = self.alt_damage
	end
	
	local _oldOnAttack = self.OnAttack
	function self:OnAttack(attacker, target, projectile)
		if not self.altattackfn then
			_oldOnAttack(self, attacker, target, projectile)
		else
			if self.altattackfn ~= nil then
				self.altattackfn(self.inst, attacker, target)
			end
			
			if self.inst.components.finiteuses ~= nil then
				self.inst.components.finiteuses:Use(self.attackwear or 1)
			end
		end
		self.damage = self.base_damage
		self.altattacking = false
	end
end)]]
